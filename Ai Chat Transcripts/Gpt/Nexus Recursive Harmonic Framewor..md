```python
import numpy as np
import matplotlib.pyplot as plt

# ======================================================
# Nexus Harmonic Recursion Map
# ======================================================
# x{t+1} = xt + H * sin(2π xt (1 - xt))  (mod 1)
# The Lyapunov exponent λ(H) measures average divergence
# ======================================================

def lyapunovexponent(H, x0=0.3, numsteps=2000, discard=500):
    """Return Lyapunov exponent for a given H."""
    x = x0
    lyapsum = 0.0
    for t in range(numsteps):
        # Map update
        xnext = (x + H * np.sin(2 * np.pi * x * (1 - x))) % 1
        # Derivative of the map with respect to x
        derivative = 1 + H * 2 * np.pi * (1 - 2 * x) * np.cos(2 * np.pi * x * (1 - x))
        if t >= discard:  # skip transients
            lyapsum += np.log(abs(derivative))
        x = xnext
    return lyapsum / (numsteps - discard)

# Sweep across H values
Hvalues = np.linspace(0.1, 0.7, 600)
lyapunovvalues = [lyapunovexponent(H) for H in Hvalues]

# Plot Lyapunov exponent vs. harmonic constant
plt.figure(figsize=(10, 6))
plt.plot(Hvalues, lyapunovvalues, color='darkred')
plt.axhline(0, color='black', linestyle='--', linewidth=0.8)
plt.title("Lyapunov Exponent vs. Harmonic Constant H")
plt.xlabel("H (harmonic constant)")
plt.ylabel("Lyapunov Exponent λ(H)")
plt.grid(True, alpha=0.3)
plt.show()
```
