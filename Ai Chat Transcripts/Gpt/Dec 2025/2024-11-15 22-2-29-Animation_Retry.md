```python
import bpy

# Initialize a new Blender scene
bpy.ops.wm.read_factory_settings(use_empty=True)

emission_mat.use_nodes = True
emission_nodes = emission_mat.node_tree.nodes
emission_links = emission_mat.node_tree.links

emission_node = emission_nodes.new(type='ShaderNodeEmission')
output_node = emission_nodes.get('Material Output')
emission_node.inputs[1].default_value = 100.0  # High intensity

emission_links.new(emission_node.outputs[0], output_node.inputs[0])
singularity.data.materials.append(emission_mat)

# Add a particle system for quark-gluon plasma
bpy.ops.object.particle_system_add()
particle_system = singularity.particle_systems[0]
particle_system.name = "QuarkGluonPlasma"

# Configure the particle system
settings = particle_system.settings
settings.count = 5000
settings.lifetime = 100
settings.frame_start = 1
settings.frame_end = 50
settings.render_type = 'HALO'
settings.particle_size = 0.05
settings.physics_type = 'NEWTON'

# Add a light source for global illumination
bpy.ops.object.light_add(type='SUN', location=(0, 0, 10))
light = bpy.context.object
light.data.energy = 5

# Set the background to black
world = bpy.context.scene.world
world.use_nodes = True
world.node_tree.nodes['Background'].inputs[1].default_value = 0.0

# Add camera
bpy.ops.object.camera_add(location=(0, -5, 2))
camera = bpy.context.object
camera.rotation_euler = (1.2, 0, 0)
bpy.context.scene.camera = camera

# Save the file
bpy.ops.wm.save_as_mainfile(filepath="First_10_Seconds_Universe.blend")
```
