The issue you are experiencing with **Entity Framework (EF) migrations** generating **relationships** between inherited entities rather than **filling all base entity fields** is due to **Table Per Type (TPT) inheritance mapping** being applied.

By default, when using **inheritance** in EF Core, the **default mapping strategy is Table Per Hierarchy (TPH)**, where all derived entities are stored in a single table with a **discriminator column**.

However, your migration script explicitly sets:
```csharp
b.UseTptMappingStrategy();
```
This tells EF Core to use **Table Per Type (TPT)**, which results in **each derived entity getting its own table** that includes only the properties **specific to that entity**, while keeping a **foreign key reference** to the base table.

---

### **Solution: Use Table Per Hierarchy (TPH) Instead of TPT**
If you want **all fields from the base entity** to be present in each derived entity's table rather than using a **relationship**, you need to **switch from TPT to TPH**.

#### **Modify the ModelBuilder Configuration**
Remove `UseTptMappingStrategy()` and force **TPH strategy**, which will create a single table including all properties from the base and derived entities:
```csharp
protected override void OnModelCreating(ModelBuilder modelBuilder)
{
    modelBuilder.Entity<JobBaseEntity>()
        .UseTphMappingStrategy(); // Use Table Per Hierarchy (default)
}
```
**TPH Behavior:**
- **One table for all types**, with a **discriminator column** to distinguish each type.
- **All fields (from base and derived entities) exist in the same table**.

---

### **Alternative: Use Table Per Concrete Type (TPC)**
If you want **each derived entity to have its own table**, but you want **all base properties copied into each derived entity’s table (without relationships)**, use **TPC (Table Per Concrete Type)**:
```csharp
modelBuilder.Entity<JobBaseEntity>().UseTpcMappingStrategy();
```
**TPC Behavior:**
- **Each entity gets its own table**, including all **base entity properties**.
- **No foreign key relationships between tables**.
- **Queries involving the base entity must UNION all derived tables**.

---

### **Summary of Inheritance Strategies**
| Strategy | Tables Created | Inheritance Storage | Relationships |
|----------|--------------|------------------|--------------|
| **TPH (Table Per Hierarchy)** | **One table for all entities** | **Single table with a discriminator column** | **No relationships** |
| **TPT (Table Per Type)** (Current Setup) | **One table per entity** | **Base table stores common fields, child tables store specific fields** | **Foreign key reference to the base table** |
| **TPC (Table Per Concrete Type)** | **One table per entity** | **Each table contains all base and derived fields** | **No relationships** |

namespace Logistix.Server.Application.Factory
{
    /// <summary>
    /// Generates JWT tokens for authenticated users.
    /// Integrates with ASP.NET Identity claims and roles.
    /// </summary>
    public class JwtTokenGenerator
    {
        private readonly IConfiguration _configuration;
        private readonly UserManager<UserEntity> _userManager;

        public JwtTokenGenerator(IConfiguration configuration, UserManager<UserEntity> userManager)
        {
            _configuration = configuration ?? throw new ArgumentNullException(nameof(configuration));
            _userManager = userManager ?? throw new ArgumentNullException(nameof(userManager));
        }

        /// <summary>
        /// Generates a JWT token for the authenticated user.
        /// </summary>
        public async Task<string> GenerateJwtToken(UserEntity user, DeviceEntity device)
        {
            if (user == null) throw new ArgumentNullException(nameof(user));
            if (device == null) throw new ArgumentNullException(nameof(device));

            // Read JWT settings from configuration
            var jwtConfig = _configuration.GetSection("Jwt");
            var key = jwtConfig["Key"];
            var audience = jwtConfig["Audience"];
            var issuer = jwtConfig["Issuer"];

            if (string.IsNullOrWhiteSpace(key) || string.IsNullOrWhiteSpace(audience) || string.IsNullOrWhiteSpace(issuer))
            {
                throw new InvalidOperationException("JWT configuration settings are missing.");
            }

            var signingKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(key));
            var creds = new SigningCredentials(signingKey, SecurityAlgorithms.HmacSha256);

            // Fetch user roles from Identity
            var roles = await _userManager.GetRolesAsync(user);
            var roleClaims = roles.Select(role => new Claim(ClaimTypes.Role, role)).ToList();

            // Fetch user claims from Identity
            var userClaims = await _userManager.GetClaimsAsync(user);

            // Define claims for the token
            var claims = new List<Claim>
            {
                new Claim(JwtRegisteredClaimNames.Sub, user.Id.ToString()),
                new Claim(JwtRegisteredClaimNames.Email, user.Email ?? ""),
                new Claim("deviceId", device.DeviceGUID.ToString()),
                new Claim("userName", user.UserName ?? ""),
                new Claim("userNickname", user.UserNickname ?? ""),
                new Claim("userAccessGroups", user.AccessGroup ?? ""),
                new Claim("signalRHubGroups", JsonSerializer.Serialize(device.SignalRGroups)),
                new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
            };

            // Merge role claims and Identity claims
            claims.AddRange(roleClaims);
            claims.AddRange(userClaims);

            // Create the token descriptor
            var tokenDescriptor = new SecurityTokenDescriptor
            {
                Subject = new ClaimsIdentity(claims),
                Expires = DateTime.UtcNow.AddDays(1),
                SigningCredentials = creds,
                Audience = audience,
                Issuer = issuer
            };

            var tokenHandler = new JwtSecurityTokenHandler();
            var token = tokenHandler.CreateToken(tokenDescriptor);

            return tokenHandler.WriteToken(token);
        }
    }
}

            // Read JWT settings from configuration
            var jwtConfig = _configuration.GetSection("Jwt");
            var key = jwtConfig["Key"];
            var audience = jwtConfig["Audience"];

            var roles = await _userManager.GetRolesAsync(user) ?? new List<string>();
            var roleClaims = roles.Select(role => new Claim(ClaimTypes.Role, role)).ToList();

            var userClaims = await _userManager.GetClaimsAsync(user) ?? new List<Claim>();

                new Claim("userName", user.UserName ?? string.Empty),
                new Claim("userNickname", user.UserNickname ?? string.Empty),
                new Claim("signalRHubGroups", signalRGroupsJson), // Properly stored as JSON
                new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
            };

            claims.AddRange(userClaims);

                Issuer = issuer
