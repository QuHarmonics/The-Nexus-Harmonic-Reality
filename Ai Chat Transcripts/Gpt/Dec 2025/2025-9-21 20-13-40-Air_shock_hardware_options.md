## 🧠 What goes into building your own oscilloscope (USB or Thunderbolt)

Here’s a high-level architecture:

```
[Analog Front End] → [ADC] → [FPGA / Microcontroller] → [USB/Thunderbolt Interface] → [PC Software]
```

| Block | Description | Notes |
|------|-------------|-------|
| **Analog Front End (AFE)** | Buffers, filters, attenuators, over-voltage protection | Needs to handle high input voltages (±20V or more), anti-aliasing, clamping, etc. |
| **ADC (Analog-to-Digital Converter)** | Samples the signal | Choose based on resolution (8/10/12/14/16-bit) and sample rate (≥50 MS/s for most oscilloscope apps) |
| **Processor (FPGA or fast MCU)** | Reads ADC data, handles buffering, preprocessing | FPGA = high-speed parallel; MCU = easier dev but limited |
| **Interface (USB or Thunderbolt)** | Sends data to host PC | USB 3.0 easiest, Thunderbolt = complex but powerful |
| **PC Software** | Visualization + control | Write in Python (PyQt), C#, Electron — plot waveforms in real time |

- **Zynq SoC (FPGA + ARM)**: Serious power, can run Linux and stream over USB or Ethernet

### 🔌 Interface
- **USB 3.0** via Cypress FX3 (streaming up to 320 MB/s)
- **Thunderbolt**: You’ll need Intel/Apple-certified controllers. Not hobby-friendly yet. PCIe over Thunderbolt is possible but complex.

def sha256_bytes(b:bytes) -> bytes:

def hash_to_scalar_0_1(digest:bytes, take_bits:int=128) -> float:
    """Normalize first `take_bits` of SHA-256 digest to [0,1)."""
    n_bytes = (take_bits+7)//8
    v = int.from_bytes(digest[:n_bytes], 'big')
    scale = 1 << (8*n_bytes)
    return v / scale

HEXMAP = "0123456789ABCDEF"

def _series(j:int, n:int) -> float:
    s = 0.0
    for k in range(n+1):
        s = (s + pow(16, n-k, 8*k + j) / (8*k + j)) % 1.0
    return s

def _series_tail(j:int, n:int, terms:int=25) -> float:
    t = 0.0
    for k in range(n+1, n+1+terms):
        t += pow(16.0, n - k) / (8*k + j)
    return t

def bbp_pi_hex_digit(n:int) -> int:
    x = 4.0*(_series(1,n) + _series_tail(1,n)) \
      - 2.0*(_series(4,n) + _series_tail(4,n)) \
      - 1.0*(_series(5,n) + _series_tail(5,n)) \
      - 1.0*(_series(6,n) + _series_tail(6,n))
    x = x % 1.0
    return int(16.0 * x)

def bbp_hex_window(start:int, span:int) -> str:
    return ''.join(HEXMAP[bbp_pi_hex_digit(start+i)] for i in range(span))

def sha256_bytes(b:bytes) -> bytes:
    return hashlib.sha256(b).digest()

def hash_to_scalar_0_1(digest:bytes, take_bits:int=128) -> float:
    """Normalize first `take_bits` of SHA-256 digest to [0,1)."""
    n_bytes = (take_bits + 7) // 8
    v = int.from_bytes(digest[:n_bytes], 'big')
    scale = 1 << (8 * n_bytes)
    return v / scale

# -----------------------------
# BBP π (hex) — true random access
# -----------------------------
HEXMAP = "0123456789ABCDEF"

def _series(j:int, n:int) -> float:
    s = 0.0
    for k in range(n+1):
        s = (s + pow(16, n-k, 8*k + j) / (8*k + j)) % 1.0
    return s

def _series_tail(j:int, n:int, terms:int=25) -> float:
    t = 0.0
    for k in range(n+1, n+1+terms):
        t += pow(16.0, n - k) / (8*k + j)
    return t

def bbp_pi_hex_digit(n:int) -> int:
    # π = sum_k 16^{-k} * (4/(8k+1) - 2/(8k+4) - 1/(8k+5) - 1/(8k+6))
    x = 4.0*(_series(1,n) + _series_tail(1,n)) \
      - 2.0*(_series(4,n) + _series_tail(4,n)) \
      - 1.0*(_series(5,n) + _series_tail(5,n)) \
      - 1.0*(_series(6,n) + _series_tail(6,n))
    x %= 1.0
    return int(16.0 * x)

def bbp_hex_window(start:int, span:int=PI_HEX_SPAN) -> str:
    return ''.join(HEXMAP[bbp_pi_hex_digit(start+i)] for i in range(span))

def twin_primes(limit:int) -> List[Tuple[int,int,int]]:

@dataclass
class TwinPoint:
    p:int; q:int; m:int
    s:float; hprox:float
    collapse_pi9:int   # 1 if |s - π/9| ≤ eps
    collapse_rnd:int   # 1 if |s - 0.35| ≤ eps (optional lane)

def compute_dataset(limit:int,
                    take_bits:int,
                    H_target:float,
                    eps:float,
                    rounded_lane:bool=True,
                    limit_twins:Optional[int]=None) -> List[TwinPoint]:
    twins = twin_primes(limit)
    if limit_twins is not None:
        twins = twins[:limit_twins]
    data = []
    for p,q,m in twins:
        s = hash_to_scalar_0_1(sha256_bytes(str(m).encode()), take_bits=take_bits)
        dist_pi9 = abs(s - H_target)
        hprox    = 1.0 - min(dist_pi9 / 0.5, 1.0)
        coll_pi9 = 1 if dist_pi9 <= eps else 0
        coll_rnd = 1 if rounded_lane and abs(s - H_ROUNDED) <= eps else 0
        data.append(TwinPoint(p,q,m,s,hprox,coll_pi9,coll_rnd))
    return data

def spectrum_view(df:pd.DataFrame) -> go.Figure:
    s = df.sort_values('m')['s'].to_numpy()
    s = s - s.mean()
    n = len(s)
    fig = go.Figure()
    if n >= 8:
        window = np.hanning(n)
        S = np.fft.rfft(s * window)
        freq = np.fft.rfftfreq(n, d=1.0)
        psd = (S * np.conj(S)).real
        fig.add_trace(go.Scatter(x=freq[1:], y=psd[1:], mode='lines', name='|FFT|^2', line=dict(color='darkgreen')))
    fig.update_layout(title="Power spectrum of s (ordered by midpoint)",
                      xaxis_title="normalized frequency", yaxis_title="power",
                      template="plotly_white", height=260)
    return fig

def _series(j:int, n:int) -> float:
    s = 0.0
    for k in range(n+1):
        s = (s + pow(16, n-k, 8*k + j) / (8*k + j)) % 1.0
    return s

def _series_tail(j:int, n:int, terms:int=25) -> float:
    t = 0.0
    for k in range(n+1, n+1+terms):
        t += pow(16.0, n - k) / (8*k + j)
    return t

      - 1.0*(_series(5,n) + _series_tail(5,n)) \
      - 1.0*(_series(6,n) + _series_tail(6,n))

    return int(16.0 * x)

def enrich_stats(s:np.ndarray, H:float, eps:float) -> Dict[str, float]:
    """
    Binomial enrichment test for hits = sum(|s-H|<=eps).
    Null: s ~ Uniform(0,1), p0 = probability(|s-H|<=eps) = width after clipping.
    """
    n = len(s)
    if n == 0:
        return {"n":0,"hits":0,"p0":0.0,"expected":0.0,"z":0.0,"p_value":1.0}
    # exact coverage (clip near edges)
    lo, hi = max(0.0, H-eps), min(1.0, H+eps)
    p0 = max(0.0, hi - lo)
    hits = int(np.sum((s >= lo) & (s <= hi)))
    expected = n * p0
    var = n * p0 * (1 - p0)
    if var <= 0:
        z = 0.0
        pval = 1.0 if hits == expected else 0.0
    else:
        z = (hits - expected) / math.sqrt(var)
        # two-sided normal approximation
        from math import erf, sqrt
        pval = 2.0 * (1.0 - 0.5*(1+erf(abs(z)/math.sqrt(2))))
    return {"n":n, "hits":hits, "p0":p0, "expected":expected, "z":z, "p_value":pval}

        stats = enrich_stats(s, H, eps)
        rows.append({"H":H, "z":stats["z"], "hits":stats["hits"], "p0":stats["p0"]})
    return pd.DataFrame(rows)

def shuffle_null(s:np.ndarray, H:float, eps:float, trials:int=200) -> Dict[str, float]:
    """
    Permutation/shuffle null: shuffle the s values and recompute hits
    (fixed H and eps). We only need mean & std of hits under shuffles.
    """
    n = len(s)
    lo, hi = max(0.0, H-eps), min(1.0, H+eps)
    hits=[]
    for _ in range(trials):
        s2 = np.random.permutation(s)
        hits.append(int(np.sum((s2 >= lo) & (s2 <= hi))))
    hits = np.asarray(hits, dtype=float)
    return {"mean": float(hits.mean()), "std": float(hits.std() + 1e-12)}

def compute(n_boot, n_clicks, maxN, method, takeBits, eps, roundedLane, sweepStep):
    maxN = int(maxN or MAX_N_DEFAULT)
    takeBits = int(takeBits or TAKE_BITS_DEF)
    eps = float(eps or COLL_EPS_DEF)
    sweepStep = float(sweepStep or SWEEP_STEP)
    show_r = bool(roundedLane and ("on" in roundedLane))

    if (method == "BLAKE3") and (not HAVE_BLAKE3):
        return None, "BLAKE3 selected but blake3 package not installed.", "Install with: pip install blake3"

    df, twins = compute_dataset(limit=maxN, take_bits=takeBits, method=method,
                                eps=eps, rounded_lane=show_r, limit_twins=None)

    # Stats @ π/9
    s = df['s'].to_numpy()
    stats = enrich_stats(s, H_DEFAULT, eps)
    shuf  = shuffle_null(s, H_DEFAULT, eps, trials=200)

    # Optional rounded lane
    stats_r = enrich_stats(s, H_ROUNDED, eps) if show_r else None

    # H sweep
    sw = sweep_H(s, eps, sweepStep)
    # Peak
    kmax = int(np.argmax(sw['z'].to_numpy()))
    H_peak, z_peak = float(sw['H'].iloc[kmax]), float(sw['z'].iloc[kmax])

    # Report text
    rep = []
    rep.append(f"Residue: {method}, N twins={len(df)}  (SHA/BLAKE bits={takeBits}, ε={eps})")
    rep.append(f"π/9 test: hits={stats['hits']}  expected={stats['expected']:.2f}  p0={stats['p0']:.3f}  z={stats['z']:.3f}  p≈{stats['p_value']:.3g}")
    rep.append(f"Shuffle null (hits @ π/9): mean={shuf['mean']:.2f}  std={shuf['std']:.2f}")
    if stats_r is not None:
        rep.append(f"0.35 lane: hits={stats_r['hits']}  expected={stats_r['expected']:.2f}  z={stats_r['z']:.3f}  p≈{stats_r['p_value']:.3g}")
    rep.append(f"H sweep peak: H*={H_peak:.3f}  z*={z_peak:.3f}  (π/9≈{H_DEFAULT:.6f}, 0.35)")
    report_text = "\n".join(rep)

    status = (f"Computed {len(df)} twin-prime events up to N={maxN} — "
              f"Residue={method}, SHA/BLAKE bits={takeBits}, ε={eps}, sweepStep={sweepStep}. "
              f"Collapse hits @ π/9: {stats['hits']}")

    # Collapse table rows
    rows=[]
    for _,r in df.iterrows():
        lane = ("π/9" if r["collapse_pi9"]==1 else ("0.35" if (show_r and r["collapse_rnd"]==1) else ""))
        if lane:
            rows.append({"p":int(r["p"]), "q":int(r["q"]), "m":int(r["m"]),
                         "s":round(float(r["s"]), 9), "Hprox":round(float(r["hprox"]), 6),
                         "lane":lane})

    store = {"frame": df.to_json(orient="split"),
             "rows": rows, "eps": eps, "showRounded": show_r,
             "method": method, "sweepStep": sweepStep}

    return store, status, report_text

pip install dash jupyter-dash plotly numpy sympy mpmath pandas
