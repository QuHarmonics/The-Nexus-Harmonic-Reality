import plotly.graph_objects as go
import hashlib

def is_mirror_pair(pair):
    return pair == pair[::-1]

def stream_sha256_hex(seed, pair_count=1000):
    data = seed.encode()
    hex_pairs = []
    while len(hex_pairs) < pair_count:
        hash_bytes = hashlib.sha256(data).hexdigest()
        for i in range(0, len(hash_bytes)-1, 2):
            hex_pair = hash_bytes[i:i+2]
            if len(hex_pair) == 2:
                hex_pairs.append(hex_pair)
            if len(hex_pairs) >= pair_count:
                break
        data = hash_bytes.encode()
    return hex_pairs

def generate_hex_mirror_spiral(source='sha256', pair_count=2000):
    if source == 'sha256':
