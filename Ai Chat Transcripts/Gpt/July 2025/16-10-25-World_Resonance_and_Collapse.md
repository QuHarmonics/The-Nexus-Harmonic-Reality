## 🐍 Python: Merge `.md` Files in a Folder

Here's a clean, production-ready Python script:

```python
import os

def merge_markdown_files(input_folder, output_file, sort_by='name'):
    # Get list of .md files
    files = [f for f in os.listdir(input_folder) if f.endswith('.md')]

    # Optional sorting
    if sort_by == 'name':
        files.sort()
    elif sort_by == 'modified':
        files.sort(key=lambda x: os.path.getmtime(os.path.join(input_folder, x)))

    with open(output_file, 'w', encoding='utf-8') as outfile:
        for filename in files:
            filepath = os.path.join(input_folder, filename)
            with open(filepath, 'r', encoding='utf-8') as infile:
                outfile.write(f"\n\n<!-- BEGIN {filename} -->\n\n")
                outfile.write(infile.read())
                outfile.write(f"\n\n<!-- END {filename} -->\n\n")

    print(f"Merged {len(files)} files into {output_file}")

# Usage
merge_markdown_files("path/to/folder", "merged_output.md", sort_by='name')
```

---

### 🔧 Options:
- `sort_by='name'`: alphabetically (default)
- `sort_by='modified'`: by last modified time

## ✅ Updated Python Code — Safe Markdown Merge with Encoding Fallback

Here’s an improved version that **tries UTF-8**, and **falls back** to `latin1` (or anything else you define):

```python
import os

def merge_markdown_files(input_folder, output_file, sort_by='name'):
    files = [f for f in os.listdir(input_folder) if f.endswith('.md')]

    # Optional sorting
    if sort_by == 'name':
        files.sort()
    elif sort_by == 'modified':
        files.sort(key=lambda x: os.path.getmtime(os.path.join(input_folder, x)))

    with open(output_file, 'w', encoding='utf-8') as outfile:
        for filename in files:
            filepath = os.path.join(input_folder, filename)
            try:
                with open(filepath, 'r', encoding='utf-8') as infile:
                    content = infile.read()
            except UnicodeDecodeError:
                # Fallback for incompatible encodings (like Windows-1252 or Latin-1)
                with open(filepath, 'r', encoding='latin1') as infile:
                    content = infile.read()

            outfile.write(f"\n\n<!-- BEGIN {filename} -->\n\n")
            outfile.write(content)
            outfile.write(f"\n\n<!-- END {filename} -->\n\n")

    print(f"Merged {len(files)} files into {output_file}")

# Usage
merge_markdown_files("D:\\Nexus\\Obsidian Valut\\Nexus3\\", "merged_output.md", sort_by='name')
```

---

### 🔧 Explanation:
- First it tries to read each file as UTF-8.
- If it fails, it **falls back to Latin-1** (which is very permissive and handles most Windows files safely).
- You can also replace `'latin1'` with `'windows-1252'` if needed.
