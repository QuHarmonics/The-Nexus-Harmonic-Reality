            """Compute curvature Δ² of path through π-space."""
            if len(positions) < 3:
                return 0.0
            
            curvatures = []
            for i in range(1, len(positions) - 1):
                Δ1 = positions[i] - positions[i-1]
                Δ2 = positions[i+1] - positions[i]
                curvatures.append(abs(Δ2 - Δ1))
            
            return np.mean(curvatures) if curvatures else 0.0
        
        def activateresonancechamber(self, hashhex: str, maxcycles: int = MAXRESONANCECYCLES):
            """Activate resonance chamber for hash unfolding."""
            print(f"Activating resonance chamber for hash: {hashhex[:16]}...")
            
            hashbytes = bytes.fromhex(hashhex)
            phase = self.hashtophase(hashhex)
            resonancepoints = []
            
            for cycle in tqdm(range(maxcycles)):
                # Position in π-field based on current phase
                pipos = int(phase * len(self.pi.digits))
                pivalue = self.pi.digits[pipos % len(self.pi.digits)]
                
                # Current hash byte (cycling through hash)
                hashidx = cycle % len(hashbytes)
                hashbyte = hashbytes[hashidx]
                
                # Compute interference
                interference = self.computeinterference(phase, pivalue)
                
                # Check for constructive interference
                if interference > 0.85:  # Strong constructive interference
                    resonancepoint = ResonancePoint(
                        piposition=pipos,
                        hashbyte=hashbyte,
                        interference=interference,
                        phase=phase,
                        curvature=self.computecurvature([rp.piposition for rp in resonancepoints[-10:]])
                    )
                    resonancepoints.append(resonancepoint)
                    
                    if VERBOSE and len(resonancepoints) % 100 == 0:
                        print(f"Cycle {cycle}: Found {len(resonancepoints)} resonance points")
                
                # Adjust phase for next cycle (harmonic navigation)
                phase = (phase + HTARGET + (pivalue / 1000.0)) % 1.0
                
                # Record harmonic history
                self.harmonichistory.append(phase)
                
                # Early termination if we have enough resonance points
                if len(resonancepoints) >= 256:  # Enough for message reconstruction
                    print(f"Found {len(resonancepoints)} resonance points, proceeding to unfold")
                    break
            
            self.resonancechamber = resonancepoints
            return resonancepoints
        
        def tracegeodesic(self) -> List[int]:
            """Trace minimal curvature path through resonance points."""
            if not self.resonancechamber:
                return []
            
            print("Tracing geodesic through π-space...")
            
            # Group resonance points by π position clusters
            clusters = defaultdict(list)
            for point in self.resonancechamber:
                clusteridx = point.piposition // 64  # 64-step clusters
                clusters[clusteridx].append(point)
            
            # Select best point from each cluster (highest interference)
            geodesicpoints = []
            for clusteridx in sorted(clusters.keys()):
                clusterpoints = clusters[clusteridx]
                bestpoint = max(clusterpoints, key=lambda p: p.interference)
                geodesicpoints.append(bestpoint)
            
            # Extract π positions along geodesic
            geodesicpath = [point.piposition for point in geodesicpoints]
            
            # Smooth path (moving average)
            if len(geodesicpath) >= 3:
                smoothed = []
                for i in range(len(geodesicpath)):
                    window = geodesicpath[max(0, i-1):min(len(geodesicpath), i+2)]
                    smoothed.append(int(np.mean(window)))
                geodesicpath = smoothed
            
            return geodesicpath
        
        def extractmessage(self, geodesic: List[int]) -> str:
            """Extract message from geodesic path."""
            print("Extracting message from geodesic...")
            
            if not geodesic:
                return ""
            
            # Convert π positions to bytes
            messagebytes = bytearray()
            bytebuffer = 0
            bitcount = 0
            
            for pos in geodesic:
                # Get π digit at position
                pidigit = self.pi.digits[pos % len(self.pi.digits)]
                
                # Build bytes from π digits (8 bits per byte)
                for bitpos in range(4):  # π digits are 4 bits (0-15)
                    bit = (pidigit >> bitpos) & 1
                    bytebuffer = (bytebuffer << 1) | bit
                    bitcount += 1
                    
                    if bitcount == 8:
                        messagebytes.append(bytebuffer)
                        bytebuffer = 0
                        bitcount = 0
            
            # Add any partial byte
            if bitcount > 0:
                messagebytes.append(bytebuffer << (8 - bitcount))
            
            # Try to decode as UTF-8
            try:
                message = messagebytes.decode('utf-8')
                # Filter for printable characters
                message = ''.join(c for c in message if 32 <= ord(c) <= 126 or c in '\n\r\t')
                return message
            except UnicodeDecodeError:
                # Return as hex if not valid UTF-8
                return f"[RAW HEX] {messagebytes.hex()}"

    def sha256hash(message: str) -> str:
        """Compute SHA-256 hash of message."""
        return hashlib.sha256(message.encode('utf-8')).hexdigest()
    
    def batchunfold(hashes: List[str], maxworkers: int = 4):
        """Batch unfold multiple hashes."""
        from concurrent.futures import ThreadPoolExecutor, ascompleted
        
        pifield = PiField()
        results = []
        
        with ThreadPoolExecutor(maxworkers=maxworkers) as executor:
            futuretohash = {executor.submit(SHAUnfolder(pifield).unfold, h): h for h in hashes}
            
            for future in tqdm(ascompleted(futuretohash), total=len(hashes), desc="Unfolding"):
                hashval = futuretohash[future]
                try:
                    result = future.result()
                    results.append((hashval, result))
                except Exception as e:
                    results.append((hashval, f"ERROR: {e}"))
        
        return results
