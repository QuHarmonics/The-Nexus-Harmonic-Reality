    @dataclass
    class Method:
        """A method m: X → O (measurement/query)"""
        name: str
        implementation: Callable[[np.ndarray], Any]
        outputspace: str
        
        def call(self, x: np.ndarray) -> Any:
            return self.implementation(x)
    
    @dataclass
    class Field:
        """
        A field F = (MF, IF, VF) where:
        - MF: set of admissible methods
        - IF: interpreter map X → VF
        - VF: value domain for this field
        """
        name: str
        methods: Dict[str, Method]
        interpreter: Callable[[np.ndarray], np.ndarray]  # IF
        valuedomain: str
        interfacerequirements: Set[str] = field(defaultfactory=set)
        
        def observe(self, x: SubstratePoint) -> Tuple[np.ndarray, Dict[str, Any]]:
            """Apply IF to get value vF(x) and behavioral signature σF(x)"""
            value = self.interpreter(x.rawdata)
            
            # Compute behavioral signature
            signature = {}
            for methodname, method in self.methods.items():
                try:
                    signature[methodname] = method(x.rawdata)
                except:
                    signature[methodname] = None
            
            return value, signature
        
        def equivalenceclass(self, x: SubstratePoint, 
                             substrate: 'TypelessSubstrate') -> Set[str]:
            """
            Compute [x]F = {y ∈ X: σF(x) = σF(y)}
            Returns set of point IDs in the same equivalence class
            """
            , sigx = self.observe(x)
            equivalent = set()
            
            for yid, y in substrate.points.items():
                , sigy = self.observe(y)
                if self.signaturesequal(sigx, sigy):
                    equivalent.add(yid)
            
            return equivalent
        
        def signaturesequal(self, sig1: Dict, sig2: Dict) -> bool:
            """Check if two behavioral signatures are equivalent"""
            # For simplicity, compare string representations
            # In a full implementation, would use proper equality for each type
            return str(sig1) == str(sig2)
        
        def satisfiestypespec(self, x: SubstratePoint, 
                               typespec: Dict[str, Any]) -> bool:
            """
            Check if σF(x) ∈ T (type specification)
            typespec is a dict: {methodname: allowedvaluesorconstraints}
            """
            , signature = self.observe(x)
            
            for methodname, constraint in typespec.items():
                if methodname not in signature:
                    return False
                
                value = signature[methodname]
                if isinstance(constraint, (list, tuple, set)):
                    if value not in constraint:
                        return False
                elif callable(constraint):
                    if not constraint(value):
                        return False
                else:
                    if value != constraint:
                        return False
            
            return True

    @dataclass
    class Observation:
        """Result of observing x under field F with interface A"""
        xid: str
        field: Field
        interface: Set[str]  # A ⊆ MF
        value: np.ndarray  # IF(x)
        behavioralsignature: Dict[str, Any]  # σF(x) restricted to A
        typeequivalenceclass: Set[str]  # [x]F
        projection: Callable  # Ψ{F,A} used
        
        def gettension(self, substrate: 'TypelessSubstrate') -> float:
            """Compute tension Θ(x; F, A)"""
            # Tension = incompatibility between interface demands and realizability
            # Simplified: variance in method outputs
            
            outputs = []
            for methodname in self.interface:
                if methodname in self.behavioralsignature:
                    val = self.behavioralsignature[methodname]
                    if isinstance(val, (int, float)):
                        outputs.append(val)
                    elif isinstance(val, dict) and 'score' in val:
                        outputs.append(val['score'])
            
            if len(outputs) < 2:
                return 0.0
            
            return np.var(outputs)
        
        def getrestprobability(self, beta: float = 1.0) -> float:
            """prest(x; F, A) ∝ exp(-β|Δ|)"""
            # Δ is magnitude of change this observation would induce
            # Simplified: use tension as proxy for |Δ|
            tension = self.gettension(None)  # We'll compute properly later
            return np.exp(-beta * tension)
    
    class ObservationOperator:
        """Implements observation with reflection and DI"""
        
        def init(self, substrate: TypelessSubstrate):
            self.substrate = substrate
            self.observationcache: Dict[Tuple[str, str, frozenset], Observation] = {}
        
        def observe(self, xid: str, field: Field, 
                    interface: Optional[Set[str]] = None) -> Observation:
            """Perform observation: reflection + DI"""
            
            cachekey = (xid, field.name, frozenset(interface or set()))
            if cachekey in self.observationcache:
                return self.observationcache[cachekey]
            
            x = self.substrate.points[xid]
            
            # Step 1: Reflection ρF(x)
            value, fullsignature = field.observe(x)
            
            # Step 2: DI - select interface A (if not specified, use all methods)
            if interface is None:
                interface = set(field.methods.keys())
            
            # Restrict signature to interface
            restrictedsignature = {k: v for k, v in fullsignature.items() 
                                  if k in interface}
            
            # Step 3: Type equivalence class [x]F
            typeclass = field.equivalenceclass(x, self.substrate)
            
            # Step 4: Projection Ψ{F,A}
            def projection(data: np.ndarray) -> np.ndarray:
                # Simplified projection: just use the interpreter
                return field.interpreter(data)
            
            # Create observation
            obs = Observation(
                xid=xid,
                field=field,
                interface=interface,
                value=value,
                behavioralsignature=restrictedsignature,
                typeequivalenceclass=typeclass,
                projection=projection
            )
            
            self.observationcache[cachekey] = obs
            return obs
        
        def transform(self, xid: str, fromfield: Field, tofield: Field,
                     frominterface: Set[str], tointerface: Set[str]) -> Observation:
            """
            Transform: IF(x) → transport → IG(x')
            Returns observation in target field
            """
            # Observe in source field
            sourceobs = self.observe(xid, fromfield, frominterface)
            
            # Transport: try to find x' that has similar behavior in target field
            # For simplicity, we'll use the original x and observe in target field
            targetobs = self.observe(xid, tofield, tointerface)
            
            # Compute transport error
            transporterror = np.linalg.norm(
                sourceobs.value - self.adaptvalue(sourceobs.value, fromfield, tofield)
            )
            
            # Modify observation to include transport info
            targetobs.metadata = {"transporterror": transporterror}
            
            return targetobs
        
        def adaptvalue(self, value: np.ndarray, fromfield: Field, 
                        tofield: Field) -> np.ndarray:
            """Simple value adapter between fields"""
            # Just pass through for now - real implementation would have proper adapter
            return value

    def computeomega(field1: Field, field2: Field, 
                     substrate: TypelessSubstrate,
                     observationop: ObservationOperator) -> float:
        """
        Ω(F, G) = measure of incompatibility between fields
        
        When Ω ≠ 0, no consistent joint refinement of ∼F and ∼G exists
        """
        
        # Collect all type equivalence classes
        typeclassesF = []
        typeclassesG = []
        
        for xid in substrate.points:
            obsF = observationop.observe(xid, field1)
            obsG = observationop.observe(xid, field2)
            
            typeclassesF.append(frozenset(obsF.typeequivalenceclass))
            typeclassesG.append(frozenset(obsG.typeequivalenceclass))
        
        # Check if partitions can be jointly refined
        # Two partitions are compatible if every intersection of their
        # equivalence classes is either empty or a union of classes from both
        
        conflicts = 0
        total = 0
        
        for i, classF in enumerate(typeclassesF):
            for j, classG in enumerate(typeclassesG):
                if i == j:
                    continue
                
                intersection = classF.intersection(classG)
                
                # If intersection is non-empty and not equal to either class,
                # and classes are not nested, we have a conflict
                if intersection and len(intersection) > 0:
                    if (intersection != classF and intersection != classG and
                        classF != classG):
                        # Check if intersection is a union of classes
                        # For simplicity, count as conflict
                        conflicts += 1
                total += 1
        
        omega = conflicts / max(total, 1)
        return omega

    class PsiCollapseOptimizer:
        """Implements Ψ-collapse as recursive interface negotiation"""
        
        def init(self, substrate: TypelessSubstrate, fields: List[Field]):
            self.substrate = substrate
            self.fields = {f.name: f for f in fields}
            self.obsop = ObservationOperator(substrate)
            
            # Tension functional parameters
            self.gamma = 0.1  # Decay rate for tension
            self.beta = 1.0   # Inverse temperature for rest probability
            self.learningrate = 0.01
            
        def tensionfunctional(self, xid: str, fieldname: str, 
                              interface: Set[str]) -> float:
            """Θ(x; F, A) - measures incompatibility"""
            field = self.fields[fieldname]
            obs = self.obsop.observe(xid, field, interface)
            return obs.gettension(self.substrate)
        
        def updateinterface(self, xid: str, fieldname: str,
                            currentinterface: Set[str],
                            step: int = 0) -> Set[str]:
            """
            Samson-style update: modify interface to reduce tension
            
            Δt ∝ -∇A Θ(x; F, At)
            """
            field = self.fields[fieldname]
            currenttension = self.tensionfunctional(xid, fieldname, 
                                                     currentinterface)
            
            # Explore interface modifications
            allmethods = set(field.methods.keys())
            candidateinterfaces = []
            
            # Try adding/removing methods
            for method in allmethods:
                if method in currentinterface:
                    # Try removing
                    newinterface = currentinterface - {method}
                else:
                    # Try adding
                    newinterface = currentinterface | {method}
                
                newtension = self.tensionfunctional(xid, fieldname, newinterface)
                
                if newtension < currenttension:
                    candidateinterfaces.append((newinterface, newtension))
            
            if not candidateinterfaces:
                return currentinterface
            
            # Choose best candidate
            bestinterface, besttension = min(candidateinterfaces, 
                                             key=lambda x: x[1])
            
            # Apply with momentum
            if step > 0:
                # Keep some methods from previous step
                preserved = currentinterface.intersection(bestinterface)
                if len(preserved) < len(currentinterface) // 2:
                    # Too much change, be conservative
                    return currentinterface
            
            return bestinterface
        
        def optimize(self, xid: str, fieldname: str,
                    maxiter: int = 100) -> Dict[str, Any]:
            """
            Run Ψ-collapse: recursive negotiation to find stable interface
            
            Returns when Θ{t+1} ≤ Θt and Δt → 0
            """
            field = self.fields[fieldname]
            allmethods = set(field.methods.keys())
            
            # Start with all methods
            currentinterface = allmethods
            tensions = []
            interfaces = []
            
            for t in range(maxiter):
                currenttension = self.tensionfunctional(xid, fieldname, 
                                                         currentinterface)
                tensions.append(currenttension)
                interfaces.append(currentinterface.copy())
                
                # Update interface
                newinterface = self.updateinterface(xid, fieldname, 
                                                     currentinterface, t)
                
                # Check convergence
                if newinterface == currentinterface:
                    # Check if tension is stable
                    if len(tensions) > 3:
                        recenttensions = tensions[-3:]
                        if np.std(recenttensions) < 0.01:
                            break
                
                currentinterface = newinterface
            
            # Compute stability measures
            finaltension = tensions[-1] if tensions else 0
            tensionreduction = (tensions[0] - finaltension) / (tensions[0] + 1e-10)
            
            # Compute rest probability
            finalobs = self.obsop.observe(xid, field, currentinterface)
            restprob = finalobs.getrestprobability(self.beta)
            
            return {
                "optimalinterface": currentinterface,
                "finaltension": finaltension,
                "tensionreduction": tensionreduction,
                "restprobability": restprob,
                "iterations": len(tensions),
                "tensionhistory": tensions,
                "interfacehistory": interfaces
            }
        
        def nexusproximity(self, xid: str, 
                           fieldweights: Optional[Dict[str, float]] = None) -> float:
            """
            "Closeness to Nexus" = concentration of rest probability
            across fields
            """
            if fieldweights is None:
                fieldweights = {name: 1.0/len(self.fields) 
                               for name in self.fields}
            
            restprobs = []
            for fieldname, weight in fieldweights.items():
                # Optimize interface for this field
                result = self.optimize(xid, fieldname, maxiter=50)
                restprobs.append(result["restprobability"] * weight)
            
            # Nexus proximity = weighted average of rest probabilities
            nexusprox = np.mean(restprobs)
            
            # Apply Mark1 corridor constraint
            Hmark1 = np.pi / 9
            # Simple heuristic: proximity decays with distance from ideal tension
            idealtension = 0.1  # Example ideal
            actualtension = np.mean([self.tensionfunctional(xid, name, 
                                                             self.fields[name].methods.keys())
                                    for name in self.fields])
            
            mark1factor = np.exp(-abs(actualtension - idealtension) / Hmark1)
            
            return nexusprox * mark1factor

    def simulatedisassembly(data: np.ndarray) -> Dict[str, Any]:
        """Simulate disassembly of bytes"""
        if len(data) < 2:
            return {"instructions": [], "score": 0.0}

        for i in range(0, min(len(data), 8), 2):
            b1 = data[i]
            b2 = data[i+1] if i+1 < len(data) else 0
            
            # Simple opcode simulation
            opcode = b1 >> 4
            if opcode == 1:
                instructions.append(f"MOV r{opcode}, 0x{b2:02x}")
            elif opcode == 2:
                instructions.append(f"ADD r{opcode}, 0x{b2:02x}")
            elif opcode == 3:
                instructions.append(f"SUB r{opcode}, 0x{b2:02x}")
            else:
                instructions.append(f"DB 0x{b1:02x}")
        
        # Score based on plausibility as machine code
        uniquebytes = len(set(data[:8]))
        score = min(1.0, uniquebytes / 8.0)
        
        return {"instructions": instructions, "score": score}

    def checksymmetry(data: np.ndarray) -> float:
        """Check symmetry properties"""
        if len(data) < 4:
            return 0.0
        
        # Check if first half mirrors second half
        half = len(data) // 2
        firsthalf = data[:half]
        secondhalf = data[half:2*half][::-1] if len(data) >= 2*half else data[half:]
        
        if len(firsthalf) == len(secondhalf):
            correlation = np.corrcoef(firsthalf, secondhalf)[0,1]
            return max(0, correlation)  # 0 to 1
        else:
            return 0.0
    
    def computeentropy(data: np.ndarray) -> float:
        """Compute Shannon entropy"""
        if len(data) == 0:
            return 0.0
        
        probs = np.bincount(data.astype(int) % 16, minlength=16) / len(data)
        probs = probs[probs > 0]
        return -np.sum(probs * np.log2(probs))

    def simulatequalia(data: np.ndarray) -> Dict[str, Any]:
        """Simulate qualia experience"""
        if len(data) == 0:
            return {
                "intensity": 0.0, "variability": 0.0, "valence": 0.0, 
                "coherence": 0.0, "positiveratio": 0.0, "negativeratio": 0.0, 
                "neutralratio": 0.0
            }
        
        # Map data to experiential dimensions
        sampledata = data[:min(8, len(data))]
        if len(sampledata) == 0:
            intensity = 0.0
        else:
            intensity = np.mean(sampledata) / 255.0
        
        # Variability
        if len(sampledata) > 1:
            variability = np.std(sampledata) / 128.0
        else:
            variability = 0.0
        
        # Emotional valence (positive/negative)
        midpoint = 128
        positive = np.sum(sampledata > midpoint) / max(len(sampledata), 1)
        negative = np.sum(sampledata < midpoint) / max(len(sampledata), 1)
        neutral = np.sum(sampledata == midpoint) / max(len(sampledata), 1)
        
        valence = positive - negative  # -1 to 1
        
        # Coherence (how well patterns fit together)
        angles = (data[:min(16, len(data))] / 255.0) * 2 * np.pi
        C = np.mean(np.cos(angles))
        S = np.mean(np.sin(angles))
        coherence = np.sqrt(C**2 + S**2)
        
        return {
            "intensity": float(intensity),
            "variability": float(variability),
            "valence": float(valence),
            "coherence": float(coherence),
            "positiveratio": float(positive),
            "negativeratio": float(negative),
            "neutralratio": float(neutral)
        }
    
    def checksignificance(data: np.ndarray) -> float:
        """Check if data seems significant/meaningful"""
        if len(data) < 2:
            return 0.0
        
        # Combine several measures
        entropy = computeentropy(data[:min(16, len(data))])
        normalizedentropy = entropy / 4.0  # Max for 16 symbols
        
        # Pattern strength (autocorrelation at lag 1)
        if len(data) > 1:
            try:
                correlation = np.corrcoef(data[:-1], data[1:])[0,1]
                patternstrength = max(0, abs(correlation))
            except:
                patternstrength = 0.0
        else:
            patternstrength = 0.0
        
        # Novelty (how different from uniform)
        uniformentropy = np.log2(16)  # For 16 possible values
        if uniformentropy > 0:
            novelty = 1.0 - (entropy / uniformentropy)
        else:
            novelty = 0.0
        
        # Composite score
        significance = 0.3 * normalizedentropy + 0.4 * patternstrength + 0.3 * novelty
        return min(1.0, max(0.0, significance))
    
    def measureattentionattraction(data: np.ndarray) -> float:
        """Measure how much attention this data would attract"""
        if len(data) == 0:
            return 0.0
        
        # Factors: contrast, novelty, pattern strength
        
        # Contrast (difference from mean)
        meanval = np.mean(data)
        contrast = np.mean(np.abs(data - meanval)) / 128.0
        
        # Novelty (entropy-based)
        entropy = computeentropy(data[:min(16, len(data))])
        novelty = entropy / 4.0
        
        # Pattern (repetition)
        patternscore = 0.0
        if len(data) >= 4:
            # Check for repeating patterns
            for patternlen in [2, 3, 4]:
                if len(data) >= patternlen * 2:
                    first = data[:patternlen]
                    second = data[patternlen:2*patternlen]
                    if np.arrayequal(first, second):
                        patternscore = max(patternscore, 1.0/patternlen)
        
        attention = 0.4 * contrast + 0.3 * novelty + 0.3 * patternscore
        return min(1.0, max(0.0, attention))

    # 5. OBSERVATION: Reflection ρF and DI δ{F,A}

    @dataclass
    class Observation:
        """Result of observing x under field F with interface A"""
        xid: str
        field: Field
        interface: Set[str]  # A ⊆ MF
        value: np.ndarray  # IF(x)
        behavioralsignature: Dict[str, Any]  # σF(x) restricted to A
        typeequivalenceclass: Set[str]  # [x]F
        projection: Callable  # Ψ{F,A} used
        
        def gettension(self, substrate: Optional['TypelessSubstrate'] = None) -> float:
            """Compute tension Θ(x; F, A)"""
            # Tension = incompatibility between interface demands and realizability
            # Simplified: variance in method outputs
            
            outputs = []
            for methodname in self.interface:
                if methodname in self.behavioralsignature:
                    val = self.behavioralsignature[methodname]
                    if isinstance(val, (int, float)):
                        outputs.append(val)
                    elif isinstance(val, dict) and 'score' in val:
                        outputs.append(val['score'])
            
            if len(outputs) < 2:
                return 0.0
            
            return np.var(outputs)
        
        def getrestprobability(self, beta: float = 1.0) -> float:
            """prest(x; F, A) ∝ exp(-β|Δ|)"""
            # Δ is magnitude of change this observation would induce
            # Simplified: use tension as proxy for |Δ|
            tension = self.gettension(None)
            return np.exp(-beta * tension)
    
    class ObservationOperator:
        """Implements observation with reflection and DI"""
        
        def init(self, substrate: TypelessSubstrate):
            self.substrate = substrate
            self.observationcache: Dict[Tuple[str, str, frozenset], Observation] = {}
        
        def observe(self, xid: str, field: Field, 
                    interface: Optional[Set[str]] = None) -> Observation:
            """Perform observation: reflection + DI"""
            
            cachekey = (xid, field.name, frozenset(interface or set()))
            if cachekey in self.observationcache:
                return self.observationcache[cachekey]
            
            x = self.substrate.points[xid]
            
            # Step 1: Reflection ρF(x)
            try:
                value, fullsignature = field.observe(x)
            except Exception as e:
                print(f"Error observing {xid} with {field.name}: {e}")
                value = np.zeros(3)
                fullsignature = {}
            
            # Step 2: DI - select interface A (if not specified, use all methods)
            if interface is None:
                interface = set(field.methods.keys())
            
            # Restrict signature to interface
            restrictedsignature = {k: v for k, v in fullsignature.items() 
                                  if k in interface}
            
            # Step 3: Type equivalence class [x]F
            try:
                typeclass = field.equivalenceclass(x, self.substrate)
            except:
                typeclass = {xid}
            
            # Step 4: Projection Ψ{F,A}
            def projection(data: np.ndarray) -> np.ndarray:
                try:
                    return field.interpreter(data)
                except:
                    return np.zeros(3)
            
            # Create observation
            obs = Observation(
                xid=xid,
                field=field,
                interface=interface,
                value=value,
                behavioralsignature=restrictedsignature,
                typeequivalenceclass=typeclass,
                projection=projection
            )
            
            self.observationcache[cachekey] = obs
            return obs

    # 6. Ω: MEASURE OF FIELD INCOMPATIBILITY
    # =============================================================================
    
    def computeomega(field1: Field, field2: Field, 
                     substrate: TypelessSubstrate,
                     observationop: ObservationOperator) -> float:
        """
        Ω(F, G) = measure of incompatibility between fields
        
        When Ω ≠ 0, no consistent joint refinement of ∼F and ∼G exists
        """
        
        # Collect all type equivalence classes
        typeclassesF = []
        typeclassesG = []
        
        for xid in substrate.points:
            try:
                obsF = observationop.observe(xid, field1)
                obsG = observationop.observe(xid, field2)
                
                typeclassesF.append(frozenset(obsF.typeequivalenceclass))
                typeclassesG.append(frozenset(obsG.typeequivalenceclass))
            except:
                continue
        
        if len(typeclassesF) == 0 or len(typeclassesG) == 0:
            return 1.0  # Maximum incompatibility
        
        # Check if partitions can be jointly refined
        # Two partitions are compatible if every intersection of their
        # equivalence classes is either empty or a union of classes from both
        
        conflicts = 0
        total = 0
        
        for i, classF in enumerate(typeclassesF):
            for j, classG in enumerate(typeclassesG):
                if i == j:
                    continue
                
                intersection = classF.intersection(classG)
                
                # If intersection is non-empty and not equal to either class,
                # and classes are not nested, we have a conflict
                if intersection and len(intersection) > 0:
                    if (intersection != classF and intersection != classG and
                        classF != classG):
                        # Check if intersection is a union of classes
                        # For simplicity, count as conflict
                        conflicts += 1
                total += 1
        
        omega = conflicts / max(total, 1)
        return omega

    class PsiCollapseOptimizer:
        """Implements Ψ-collapse as recursive interface negotiation"""
        
        def init(self, substrate: TypelessSubstrate, fields: List[Field]):
            self.substrate = substrate
            self.fields = {f.name: f for f in fields}
            self.obsop = ObservationOperator(substrate)
            
            # Tension functional parameters
            self.gamma = 0.1  # Decay rate for tension
            self.beta = 1.0   # Inverse temperature for rest probability
            self.learningrate = 0.01
            
        def tensionfunctional(self, xid: str, fieldname: str, 
                              interface: Set[str]) -> float:
            """Θ(x; F, A) - measures incompatibility"""
            try:
                field = self.fields[fieldname]
                obs = self.obsop.observe(xid, field, interface)
                return obs.gettension(self.substrate)
            except:
                return 1.0  # Maximum tension on error
        
        def updateinterface(self, xid: str, fieldname: str,
                            currentinterface: Set[str],
                            step: int = 0) -> Set[str]:
            """
            Samson-style update: modify interface to reduce tension
            
            Δt ∝ -∇A Θ(x; F, At)
            """
            try:
                field = self.fields[fieldname]
            except:
                return currentinterface
            
            currenttension = self.tensionfunctional(xid, fieldname, 
                                                     currentinterface)
            
            # Explore interface modifications
            allmethods = set(field.methods.keys())
            candidateinterfaces = []
            
            # Try adding/removing methods
            for method in allmethods:
                if method in currentinterface:
                    # Try removing
                    newinterface = currentinterface - {method}
                else:
                    # Try adding
                    newinterface = currentinterface | {method}
                
                newtension = self.tensionfunctional(xid, fieldname, newinterface)
                
                if newtension < currenttension:
                    candidateinterfaces.append((newinterface, newtension))
            
            if not candidateinterfaces:
                return currentinterface
            
            # Choose best candidate
            bestinterface, besttension = min(candidateinterfaces, 
                                             key=lambda x: x[1])
            
            # Apply with momentum
            if step > 0:
                # Keep some methods from previous step
                preserved = currentinterface.intersection(bestinterface)
                if len(preserved) < len(currentinterface) // 2:
                    # Too much change, be conservative
                    return currentinterface
            
            return bestinterface
        
        def optimize(self, xid: str, fieldname: str,
                    maxiter: int = 50) -> Dict[str, Any]:
            """
            Run Ψ-collapse: recursive negotiation to find stable interface
            
            Returns when Θ{t+1} ≤ Θt and Δt → 0
            """
            try:
                field = self.fields[fieldname]
            except:
                return {
                    "optimalinterface": set(),
                    "finaltension": 1.0,
                    "tensionreduction": 0.0,
                    "restprobability": 0.0,
                    "iterations": 0,
                    "tensionhistory": [],
                    "interfacehistory": []
                }
            
            allmethods = set(field.methods.keys())
            
            # Start with all methods
            currentinterface = allmethods
            tensions = []
            interfaces = []
            
            for t in range(maxiter):
                try:
                    currenttension = self.tensionfunctional(xid, fieldname, 
                                                             currentinterface)
                except:
                    currenttension = 1.0
                
                tensions.append(currenttension)
                interfaces.append(currentinterface.copy())
                
                # Update interface
                newinterface = self.updateinterface(xid, fieldname, 
                                                     currentinterface, t)
                
                # Check convergence
                if newinterface == currentinterface:
                    # Check if tension is stable
                    if len(tensions) > 3:
                        recenttensions = tensions[-3:]
                        if np.std(recenttensions) < 0.01:
                            break
                
                currentinterface = newinterface
            
            # Compute stability measures
            finaltension = tensions[-1] if tensions else 1.0
            if tensions and tensions[0] > 0:
                tensionreduction = (tensions[0] - finaltension) / tensions[0]
            else:
                tensionreduction = 0.0
            
            # Compute rest probability
            try:
                finalobs = self.obsop.observe(xid, field, currentinterface)
                restprob = finalobs.getrestprobability(self.beta)
            except:
                restprob = 0.0
            
            return {
                "optimalinterface": currentinterface,
                "finaltension": float(finaltension),
                "tensionreduction": float(tensionreduction),
                "restprobability": float(restprob),
                "iterations": len(tensions),
                "tensionhistory": [float(t) for t in tensions],
                "interfacehistory": interfaces
            }
        
        def nexusproximity(self, xid: str, 
                           fieldweights: Optional[Dict[str, float]] = None) -> float:
            """
            "Closeness to Nexus" = concentration of rest probability
            across fields
            """
            if fieldweights is None:
                fieldweights = {name: 1.0/len(self.fields) 
                               for name in self.fields}
            
            restprobs = []
            for fieldname, weight in fieldweights.items():
                # Optimize interface for this field
                result = self.optimize(xid, fieldname, maxiter=30)
                restprobs.append(result["restprobability"] * weight)
            
            # Nexus proximity = weighted average of rest probabilities
            if restprobs:
                nexusprox = np.mean(restprobs)
            else:
                nexusprox = 0.0
            
            # Apply Mark1 corridor constraint
            Hmark1 = np.pi / 9
            # Simple heuristic: proximity decays with distance from ideal tension
            idealtension = 0.1  # Example ideal
            
            # Calculate average tension
            tensions = []
            for fieldname in self.fields:
                field = self.fields[fieldname]
                try:
                    tension = self.tensionfunctional(xid, fieldname, 
                                                     set(field.methods.keys()))
                    tensions.append(tension)
                except:
                    tensions.append(1.0)
            
            if tensions:
                actualtension = np.mean(tensions)
            else:
                actualtension = 1.0
            
            mark1factor = np.exp(-abs(actualtension - idealtension) / (Hmark1 + 1e-10))
            
            return float(nexusprox * mark1factor)
