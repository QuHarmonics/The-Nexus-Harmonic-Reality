                fft = np.fft.fft(data - np.mean(data))
                freqs = np.fft.fftfreq(len(data))
                power = np.abs(fft)**2
                
                # Split into blue/red domains

                bluemask = np.abs(freqs) < cutoff
                redmask = np.abs(freqs) >= cutoff

                results['spectrum']['redenergy'] = float(np.sum(power[redmask]))
                results['spectrum']['blueratio'] = float(

        @staticmethod
        def validatestream(stream: bytes, expectedhash: str = None) -> Dict:
            """Comprehensive stream validation"""
            results = {
                'valid': True,
                'errors': [],
                'warnings': [],
                'metrics': {},
            }
            
            # Basic checks
            if len(stream) == 0:
                results['valid'] = False
                results['errors'].append("Empty stream")
            
            # Hash verification
            actualhash = hashlib.sha256(stream).hexdigest()
            if expectedhash and actualhash != expectedhash:
                results['valid'] = False
                results['errors'].append(f"Hash mismatch: {actualhash[:16]} != {expectedhash[:16]}")
            
            # Byte distribution test (chi-square)
            bytecounts = np.bincount(np.frombuffer(stream, dtype=np.uint8), minlength=256)
            expected = len(stream) / 256
            chi2 = np.sum((bytecounts - expected)**2 / expected)
            
            results['metrics']['chi2'] = chi2
            results['metrics']['uniformityp'] = 1 - np.exp(-chi2/2)  # Approximate p-value
            
            if chi2 > 300:  # Threshold for 256 degrees of freedom
                results['warnings'].append(f"Non-uniform distribution: χ²={chi2:.1f}")
            
            # Entropy check
            dist = bytecounts[bytecounts > 0] / len(stream)
            entropy = -np.sum(dist * np.log2(dist))
            results['metrics']['entropy'] = entropy
            
            if entropy < 7.9:  # Close to maximum 8.0
                results['warnings'].append(f"Suboptimal entropy: {entropy:.3f}")
            
            return results
        
        @staticmethod
        def comparestreams(stream1: bytes, stream2: bytes) -> Dict:
            """Compare two Ξₙₑₓ streams"""
            # Convert to arrays
            data1 = np.frombuffer(stream1, dtype=np.uint8)
            data2 = np.frombuffer(stream2, dtype=np.uint8)
            
            # Ensure equal length
            minlen = min(len(data1), len(data2))
            data1 = data1[:minlen]
            data2 = data2[:minlen]
            
            # Correlation
            correlation = np.corrcoef(data1, data2)[0, 1]
            
            # Hamming distance
            diff = np.sum(data1 != data2)
            hamming = diff / minlen
            
            # KL divergence
            hist1 = np.bincount(data1, minlength=256) / minlen
            hist2 = np.bincount(data2, minlength=256) / minlen
            
            # Avoid zeros for KL
            eps = 1e-10
            hist1 = hist1 + eps
            hist2 = hist2 + eps
            
            kldiv = np.sum(hist1 * np.log2(hist1 / hist2))
            
            return {
                'correlation': float(correlation),
                'hammingdistance': float(hamming),
                'kldivergence': float(kldiv),
                'bytescompared': minlen,
                'identical': bool(correlation > 0.999 and hamming < 0.001),
            }

        @staticmethod

            s = 0.0

                s += series(k)
            
            s -= int(s)
            hexdigits = []
            for  in range(numdigits):
                s *= 16
                digit = int(s)
                hexdigits.append(format(digit, 'x'))
                s -= digit
            
            return ''.join(hexdigits)

            self.mark1 = math.pi / 9  # Harmonic constant
            
            # Generate or load constants
            print("📥 Loading constants...")
            self.pihex = pisample if precision <= 1000 else NexusConstants.generatepihex(precision)
            self.ehex = esample if precision <= 1000 else NexusConstants.generateehex(precision)
            self.phihex = phisample if precision <= 1000 else NexusConstants.generatephihex(precision)
            
            # Statistics
            self.stats = {
                'blocksgenerated': 0,
                'bytesproduced': 0,
                'genlockpassed': 0,
                'genlockfailed': 0,
                'noncestested': 0,
            }
            
            print(f"✅ Engine initialized with {precision:,} hex digits")
        
        def kineticfold(self, piwindow, ewindow, phiwindow, nonce=0):
            """Kinetic folding with nonce injection"""
            # Convert hex to bytes
            pibytes = bytes.fromhex(piwindow)
            ebytes = bytes.fromhex(ewindow)
            phibytes = bytes.fromhex(phiwindow)
            
            # Ensure equal length
            minlen = min(len(pibytes), len(ebytes), len(phibytes))
            pibytes = pibytes[:minlen]
            ebytes = ebytes[:minlen]
            phibytes = phibytes[:minlen]
            
            # Nonce perturbation (golden ratio scaled)
            if nonce != 0:
                phifactor = (nonce * 1.618033988749895) % 1.0
                phibytes = bytes((b ^ int(phifactor * 255)) for b in phibytes)
            
            # Triple fold: π ⊕ e ⊕ φ(nonce)
            folded = bytes(p ^ e ^ ph for p, e, ph in zip(pibytes, ebytes, phibytes))
            
            # SHA-256 compression
            return hashlib.sha256(folded).digest()
        
        def calculatecoherence(self, block):
            """Calculate coherence score with multiple metrics"""
            if len(block) < 8:
                return 0.0
            
            data = np.frombuffer(block, dtype=np.uint8)
            
            # 1. Vector coherence (unit circle)
            angles = data / 255.0 * 2 * np.pi
            xsum = np.sum(np.cos(angles))
            ysum = np.sum(np.sin(angles))
            vectorcoherence = np.sqrt(xsum**2 + ysum**2) / len(data)
            
            # 2. Mark1 harmonic alignment
            deviations = np.minimum(
                np.abs(angles - self.mark1),
                2*np.pi - np.abs(angles - self.mark1)
            )
            harmoniccoherence = 1 - np.mean(deviations) / np.pi
            
            # 3. Byte pattern coherence (odd/even ratio)
            oddratio = np.sum(data % 2) / len(data)
            targetodd = 0.349  # Mark1 scaled
            patterncoherence = 1 - abs(oddratio - targetodd)
            
            # Combined score (weighted)
            coherence = (vectorcoherence * 0.3 + 
                        harmoniccoherence * 0.4 + 
                        patterncoherence * 0.3)
            
            return float(np.clip(coherence, 0, 1))
        
        def generateblock(self, position, windowsize=32, nonce=0, threshold=0.8):
            """Generate a single Ξₙₑₓ block"""
            # Extract windows
            piwin = self.pihex[position:position + windowsize]
            ewin = self.ehex[position:position + windowsize]
            phiwin = self.phihex[position:position + windowsize]
            
            # Ensure even length
            for win in [piwin, ewin, phiwin]:
                if len(win) % 2 != 0:
                    win = win[:-1]
            
            if not all([piwin, ewin, phiwin]):
                return None, 0.0
            
            # Generate block
            block = self.kineticfold(piwin, ewin, phiwin, nonce)
            coherence = self.calculatecoherence(block)
            
            # Genlock gate
            if coherence >= threshold:
                return block, coherence
            return None, coherence
        
        def generatestream(self, startpos=0, nonce=0, numblocks=100, 
                           windowsize=32, threshold=0.8, showprogress=True):
            """Generate Ξₙₑₓ stream"""
            stream = b""
            coherences = []
            positions = []
            
            iterator = trange(numblocks, desc=f"Ξₙₑₓ generation (nonce={nonce})") if showprogress else range(numblocks)
            
            for i in iterator:
                pos = startpos + i
                if pos + windowsize > len(self.pihex):
                    break
                
                block, coherence = self.generateblock(pos, windowsize, nonce, threshold)
                
                if block is not None:
                    stream += block
                    coherences.append(coherence)
                    positions.append(pos)
                
                self.stats['blocksgenerated'] += 1
            
            self.stats['bytesproduced'] += len(stream)
            self.stats['genlockpassed'] += len(coherences)
            self.stats['genlockfailed'] += numblocks - len(coherences)
            
            return stream, coherences, positions
        
        def analyzestream(self, stream):
            """Comprehensive stream analysis"""
            if len(stream) == 0:
                return {}
            
            data = np.frombuffer(stream, dtype=np.uint8)
            
            # Basic statistics
            stats = {
                'length': len(stream),
                'mean': float(np.mean(data)),
                'std': float(np.std(data)),
                'min': int(np.min(data)),
                'max': int(np.max(data)),
                'entropy': self.calculateentropy(data),
            }
            
            # Opcode analysis
            opcodestats = self.analyzeopcodes(data)
            stats.update(opcodestats)
            
            # Autocorrelation
            acfstats = self.autocorrelationanalysis(data)
            stats.update(acfstats)
            
            # Spectral analysis
            spectralstats = self.spectralanalysis(data)
            stats.update(spectralstats)
            
            return stats
        
        def calculateentropy(self, data):
            """Calculate Shannon entropy"""
            hist = np.bincount(data, minlength=256)
            hist = hist[hist > 0] / len(data)
            return float(-np.sum(hist * np.log2(hist)))
        
        def analyzeopcodes(self, data):
            """Analyze kinetic opcodes in stream"""
            constructive = 0
            destructive = 0
            transformative = 0
            neutral = 0
            totalenergy = 0.0
            
            for byte in data:
                high = (byte >> 4) & 0x0F
                low = byte & 0x0F
                
                for nibble in [high, low]:
                    if nibble in KINETICMAPPER:
                        op = KINETICMAPPER[nibble]
                        totalenergy += op['weight']
                        
                        if op['type'] == 'CONSTRUCTIVE':
                            constructive += 1
                        elif op['type'] == 'DESTRUCTIVE':
                            destructive += 1
                        elif op['type'] == 'TRANSFORM':
                            transformative += 1
                        else:
                            neutral += 1
            
            metabolicratio = constructive / destructive if destructive > 0 else float('inf')
            
            return {
                'constructive': constructive,
                'destructive': destructive,
                'transformative': transformative,
                'neutral': neutral,
                'totalenergy': totalenergy,
                'metabolicratio': metabolicratio,
                'expectedratio': 1.61803398875 * (math.pi/9) + 1,
            }
        
        def autocorrelationanalysis(self, data, maxlag=10):
            """Calculate autocorrelation statistics"""
            results = {}
            
            for lag in range(1, maxlag + 1):
                if lag < len(data):
                    corr = np.corrcoef(data[:-lag], data[lag:])[0, 1]
                    results[f'acflag{lag}'] = float(corr)
            
            # Double bend detection
            results['doublebend'] = results.get('acflag2', 0) < 0
            results['doublebendstrength'] = abs(results.get('acflag2', 0))
            
            return results
        
        def spectralanalysis(self, data):
            """Spectral analysis of stream"""
            if len(data) < 100:
                return {}
            
            # FFT
            fft = np.fft.fft(data - np.mean(data))
            freqs = np.fft.fftfreq(len(data))
            power = np.abs(fft)**2
            
            # Split by Mark1 harmonic
            cutoff = self.mark1 / (2 * np.pi)
            bluemask = np.abs(freqs) < cutoff
            redmask = np.abs(freqs) >= cutoff
            
            blueenergy = np.sum(power[bluemask])
            redenergy = np.sum(power[redmask])
            totalenergy = blueenergy + redenergy
            
            return {
                'blueenergy': float(blueenergy),
                'redenergy': float(redenergy),
                'blueratio': float(blueenergy / totalenergy) if totalenergy > 0 else 0,
                'spectralflatness': float(np.exp(np.mean(np.log(power[power > 0]))) / np.mean(power)),
            }

    def crosscorrelatestreams(stream1, stream2, name1="Stream 1", name2="Stream 2"):
        """Cross-correlate two Ξₙₑₓ streams"""
        data1 = np.frombuffer(stream1, dtype=np.uint8)
        data2 = np.frombuffer(stream2, dtype=np.uint8)
        
        # Normalize
        data1norm = (data1 - np.mean(data1)) / np.std(data1)
        data2norm = (data2 - np.mean(data2)) / np.std(data2)
        
        # Cross-correlation
        correlation = np.correlate(data1norm, data2norm, mode='full')
        lags = np.arange(-len(data1) + 1, len(data2))
        
        fig, axes = plt.subplots(2, 2, figsize=(14, 10))
