    using System.Collections.Generic;
    using System.IO;

    public class ConversationExtractor
    {
        public class Message
        {
            public string Role { get; set; }
            public Content Content { get; set; }
        }
    
        public class Content
        {
            public string ContentType { get; set; }
            public List<string> Parts { get; set; }

        public class ConversationNode
        {
            public string Id { get; set; }

        public static List<(string role, string content)> ExtractConversationPairs(string jsonFilePath)
        {
            var jsonString = File.ReadAllText(jsonFilePath);
            var jsonDoc = JsonDocument.Parse(jsonString);
            
            var mapping = jsonDoc.RootElement.GetProperty("mapping");
            var conversationPairs = new List<(string, string)>();
            
            // Find all nodes with actual content
            foreach (var node in mapping.EnumerateObject())
            {
                var messageElement = node.Value.GetProperty("message");
                
                if (messageElement.ValueKind == JsonValueKind.Null)
                    continue;
                    
                var role = messageElement.GetProperty("author").GetProperty("role").GetString();
                var content = messageElement.GetProperty("content");
                
                // Skip system messages and empty content
                if (role == "system" || content.ValueKind == JsonValueKind.Null)
                    continue;
                    
                var parts = content.GetProperty("parts");
                if (parts.ValueKind == JsonValueKind.Array && parts.GetArrayLength() > 0)
                {
                    var firstPart = parts[0];
                    if (firstPart.ValueKind == JsonValueKind.String)
                    {
                        var text = firstPart.GetString();
                        if (!string.IsNullOrWhiteSpace(text))
                        {
                            conversationPairs.Add((role, text));
                        }
                    }
                }
            }
            
            return conversationPairs;
        }
    }

    using System;
    using System.Collections.Generic;
    using System.IO;

    public class OpenAIConversationExtractor
    {
        public class ConversationData
        {
            public string title { get; set; }
            public double createtime { get; set; }
            public double updatetime { get; set; }
            public Dictionary<string, ConversationNode> mapping { get; set; }
            public string currentnode { get; set; }
            public string conversationid { get; set; }
        }
    
        public class ConversationNode
        {
            public string id { get; set; }
            public NodeMessage message { get; set; }
            public string parent { get; set; }
            public List<string> children { get; set; }
        }
    
        public class NodeMessage
        {
            public string id { get; set; }
            public Author author { get; set; }
            public double? createtime { get; set; }
            public MessageContent content { get; set; }
            public string status { get; set; }
        }
    
        public class Author
        {
            public string role { get; set; }
        }
    
        public class MessageContent
        {
            public string contenttype { get; set; }
            public List<string> parts { get; set; }
        }
    
        public static List<(string role, string content, double? timestamp)> ExtractConversation(string jsonFilePath)
        {
            var jsonString = File.ReadAllText(jsonFilePath);
            var conversation = JsonSerializer.Deserialize<ConversationData>(jsonString);
            
            var messages = new List<(string, string, double?)>();
            
            foreach (var node in conversation.mapping.Values)
            {
                if (node.message?.content?.parts == null || node.message.content.parts.Count == 0)
                    continue;
                    
                var role = node.message.author?.role;
                var content = node.message.content.parts[0];
                var timestamp = node.message.createtime;
                
                // Skip system messages and empty content
                if (role != "system" && !string.IsNullOrWhiteSpace(content))
                {
                    messages.Add((role, content, timestamp));
                }
            }
            
            return messages;
        }
    }
