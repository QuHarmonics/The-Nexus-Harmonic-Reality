def hill_climb(secret, n, max_q=200000):
    x = rng.integers(0, 2, n)
    q = 0
    fit = int(np.sum(x == secret)); q += 1
    while fit < n and q < max_q:
        improved = False
        for i in range(n):
            x[i] ^= 1
            f2 = int(np.sum(x == secret)); q += 1
            if f2 > fit:
                fit = f2; improved = True
            else:
                x[i] ^= 1
            if fit == n or q >= max_q:
                break
        if not improved:
            break
    return q, fit == n

def blind_search(secret, n, max_q=4_000_000):
    """flat landscape: the ONLY information is 'is this it?'"""
    target = int("".join(map(str, secret)), 2)
    N = 2**n
    order = rng.permutation(N)
    for q, cand in enumerate(order, 1):
        if cand == target:
            return q, True
        if q >= max_q:
            return q, False
    return N, True

def make_planted_3sat(n, m, sol, rng):
    """random 3-SAT clauses, each guaranteed satisfied by `sol`."""
    cl = np.zeros((m, 3), dtype=np.int32)
    sg = np.zeros((m, 3), dtype=np.int8)
    i = 0
    while i < m:
        v = rng.choice(n, 3, replace=False)
        s = rng.integers(0, 2, 3)*2 - 1                # +1 = positive literal
        lit_true = ((sol[v] == 1) & (s == 1)) | ((sol[v] == 0) & (s == -1))
        if lit_true.any():                             # satisfied by sol
            cl[i], sg[i] = v, s
            i += 1
    return cl, sg

def nsat(x, cl, sg):
    lit = np.where(sg == 1, x[cl], 1 - x[cl])
    return int(np.sum(lit.any(axis=1)))

def greedy_descend(cl, sg, n, start, max_steps=4000):
    """pure hill-climb on satisfied-clause count. Follow the gradient."""
    x = start.copy()
    cur = nsat(x, cl, sg)
    for step in range(max_steps):
        best_gain, best_i = 0, -1
        for i in range(n):
            x[i] ^= 1
            g = nsat(x, cl, sg) - cur
            x[i] ^= 1
            if g > best_gain:
                best_gain, best_i = g, i
        if best_i < 0:
            return x, cur, step, False          # LOCAL OPTIMUM. stuck.
        x[best_i] ^= 1
        cur += best_gain
        if cur == len(cl):
            return x, cur, step + 1, True       # found a satisfying assignment
    return x, cur, max_steps, False
