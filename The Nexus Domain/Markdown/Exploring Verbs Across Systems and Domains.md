def test_scale_invariance(data, n_bootstrap=1000):
    """
    Test if leak rate is invariant to uncertainty scaling
    """
    base_leak_rate = np.mean(data['leak_events'])
    
    scale_factors = np.logspace(-1, 1, 20)  # 0.1 to 10
    leak_rates = []
    confidence_intervals = []
    
    for scale in scale_factors:
        # Artificially scale uncertainty
        scaled_z = abs(data['estimate'] - data['target']) / (data['uncertainty'] * scale)
        
        # Bootstrap confidence intervals
        boot_leaks = []
        for _ in range(n_bootstrap):
            sample_idx = np.random.choice(len(scaled_z), len(scaled_z), replace=True)
            sample_z = scaled_z[sample_idx]
            # Use controller's decision rule (must be known or estimated)
            sample_leaks = sigmoid(B * (sample_z - z0)) > np.random.random(len(sample_z))
            boot_leaks.append(np.mean(sample_leaks))
        
        leak_rates.append(np.mean(boot_leaks))
        confidence_intervals.append(np.percentile(boot_leaks, [2.5, 97.5]))
    
    # Test for invariance: all leak rates within confidence of baseline?
    invariant = all(abs(lr - base_leak_rate) < 0.01 for lr in leak_rates)
    
    return {
        'scale_factors': scale_factors,
        'leak_rates': leak_rates,
        'confidence_intervals': confidence_intervals,
        'is_invariant': invariant,
        'max_deviation': max(abs(np.array(leak_rates) - base_leak_rate))
