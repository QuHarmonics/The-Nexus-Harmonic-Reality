def calculate_bit_entropy(constants):
    """Calculate bit entropy of a list of constants"""
    # Combine all constants into bytes
    all_bytes = b''
    for const in constants:
        all_bytes += const.to_bytes(4, 'big')
    
    # Calculate byte distribution
    byte_counts = np.bincount(np.frombuffer(all_bytes, dtype=np.uint8), minlength=256)
    byte_probs = byte_counts / byte_counts.sum()
    
    # Shannon entropy
    entropy = -np.sum(byte_probs[byte_probs > 0] * np.log2(byte_probs[byte_probs > 0]))
    return entropy
