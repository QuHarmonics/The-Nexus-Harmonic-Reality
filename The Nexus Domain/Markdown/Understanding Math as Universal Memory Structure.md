def pi_hex_digits(start_pos, num_digits=64):
    """
    Extract `num_digits` hexadecimal digits of π starting at `start_pos`.
    start_pos: 0‑based index of the first hex digit after the decimal point.
    Uses the BBP formula with modular exponentiation.
    """
    mpmath.mp.dps = num_digits * 4 + 20  # generous precision
    digits = []
    # For each digit position i from start_pos to start_pos+num_digits-1,
    # we need the fractional part of 16^i * π.
    # BBP: π = sum_{k=0..∞} (16^-k) * (4/(8k+1) - 2/(8k+4) - 1/(8k+5) - 1/(8k+6))
    # So 16^i * π = sum_{k=0..∞} (16^(i-k)) * term(k)
    # Split: k=0..i (integer part from 16^(i-k) mod denominator) + k=i+1..∞ (fractional)
    for i in range(start_pos, start_pos + num_digits):
        frac = mpmath.mpf(0)
        # k = 0 .. i
        for k in range(0, i + 1):
            denom1 = 8*k + 1
            denom2 = 8*k + 4
            denom3 = 8*k + 5
            denom4 = 8*k + 6
            p16 = pow(16, i - k, denom1 * denom2 * denom3 * denom4)  # not exactly; need per term
            # Better: compute term mod denominator
            term = (4 * pow(16, i - k, denom1) % denom1) / mpmath.mpf(denom1) \
                 - (2 * pow(16, i - k, denom2) % denom2) / mpmath.mpf(denom2) \
                 - (1 * pow(16, i - k, denom3) % denom3) / mpmath.mpf(denom3) \
                 - (1 * pow(16, i - k, denom4) % denom4) / mpmath.mpf(denom4)
            frac += term - mpmath.floor(term)
        # k = i+1 .. ∞
        for k in range(i + 1, i + 50):  # 50 terms is plenty; error < 16^-50
            term = (4/(8*k+1) - 2/(8*k+4) - 1/(8*k+5) - 1/(8*k+6)) / mpmath.power(16, k - i)
            frac += term
        frac = frac - mpmath.floor(frac)
        # Extract one hex digit
        frac *= 16
        digit = int(frac)
        digits.append(hex(digit)[2:].upper())
        frac -= digit
