def bbp_pi_digit(n):
    """
    Calculate a partial sum of the BBP formula that approximates π.
    This function naively sums up to the nth term.
    """
    pi_sum = 0.0
    for k in range(n + 1):
        pi_sum += (1 / (16 ** k)) * (
            4 / (8 * k + 1) -
            2 / (8 * k + 4) -
            1 / (8 * k + 5) -
            1 / (8 * k + 6)

    return pi_sum
